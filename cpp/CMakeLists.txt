cmake_minimum_required(VERSION 3.25)
project(kryvola_engine_foundation LANGUAGES CXX)
include(CTest)

add_library(kryvola_engine_contract INTERFACE)
target_compile_features(kryvola_engine_contract INTERFACE cxx_std_20)
target_include_directories(kryvola_engine_contract INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}/../contracts/spec/cpp")

add_executable(engine_foundation inspect.cpp)
target_link_libraries(engine_foundation PRIVATE kryvola_engine_contract)
add_executable(contract_boundary_test tests/contract_boundary_test.cpp)
target_link_libraries(contract_boundary_test PRIVATE kryvola_engine_contract)
foreach(target engine_foundation contract_boundary_test)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
add_test(NAME contract_boundary COMMAND contract_boundary_test)
